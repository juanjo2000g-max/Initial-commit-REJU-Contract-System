% --- Ficha de hallazgo ---------------------------------------------------
% #1 número, #2 nombre, #3 nombre sin formato (índice), #4 urgencia,
% #5 cuándo actuar, #6 responsable sugerido
\newcommand{\hallazgo}[6]{%
  \clearpage
  \phantomsection
  \addcontentsline{toc}{subsection}{Prioridad #1 · #3}%
  \begin{tcolorbox}[enhanced, sharp corners, boxrule=0pt, colback=sicharnegro,
      left=0pt, right=0pt, top=0pt, bottom=0pt, after skip=1em]
    \begin{tabular}{@{}>{\centering\arraybackslash}m{2.6cm}@{\hspace{10pt}}m{\linewidth-3.4cm}@{}}
      \cellcolor{sicharnaranja}\rule{0pt}{2.4cm}%
      \color{white}\begin{tabular}{@{}c@{}}\footnotesize\bfseries PRIORIDAD\\[2pt]
        \fontsize{38}{38}\selectfont\bfseries #1\end{tabular} &
      \color{white}{\small\itshape Hallazgo #1}\par\vspace{2pt}
      {\LARGE\bfseries #2}
    \end{tabular}
  \end{tcolorbox}
  \vspace{-0.6em}
  \noindent
  \begin{tcbraster}[raster columns=3, raster equal height, raster column skip=2mm]
    \begin{tcolorbox}[enhanced, sharp corners, boxrule=0pt, colback=rojo!10, fontupper=\small]
      {\scriptsize\bfseries\color{sichargris} URGENCIA}\par
      {\large\bfseries\color{rojo} #4}
    \end{tcolorbox}
    \begin{tcolorbox}[enhanced, sharp corners, boxrule=0pt, colback=fondogris, fontupper=\small]
      {\scriptsize\bfseries\color{sichargris} CUÁNDO ACTUAR}\par
      \textbf{#5}
    \end{tcolorbox}
    \begin{tcolorbox}[enhanced, sharp corners, boxrule=0pt, colback=fondogris, fontupper=\small]
      {\scriptsize\bfseries\color{sichargris} RESPONSABLE SUGERIDO}\par
      \textbf{#6}
    \end{tcolorbox}
  \end{tcbraster}
}
